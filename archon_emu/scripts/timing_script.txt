Start:
LowCLK; IF !LCLK GOTO Start
IDLE; CALL RCLR(4)
KEEPER; CALL EXPOSE(Expo)
KEEPER; CALL READOUT(Expo)
KEEPER; IF !Nexpo CALL ExpoDONE
KEEPER; GOTO Start
ExpoDONE:
KEEPER; Expo--
KEEPER; RETURN ExpoDONE
RCLR:
IDLE; CALL FCLR(Nclear)
KEEPER; CALL SCLR(PixSkip)
KEEPER; CALL SCLR(PixRead)
KEEPER; CALL SCLR(PixOver)
KEEPER; RETURN RCLR
FCLR:
PFA; PFA(Tpa)
PFB; PFB(Tpb)
PFC; PFC(Tpb)
PFD; PFD(Tpb)
PFE; PFE(Tpb)
PFF; PFF(Tpb)
PFG; PFG(Tpb)
PFH; PFH(Tpb)
PFI; PFI(Tpa)
KEEPER; RETURN FCLR
SCLR:
SF1;SF1(Tsf)
SF2;SF2(Tsf)
SF3;SF3(Tsf)
KEEPER; RETURN SCLR
EXPOSE:
KEEPER; IF Light OSHUT
KEEPER; CALL Integrate(ExpTime)
KEEPER; IF Light CSHUT
KEEPER; CALL Integrate(WaiTime)
KEEPER; RETURN EXPOSE
Integrate:
KEEPER; CALL IntFlush(275)
KEEPER; RETURN Integrate
IntFlush:
SR1;SR1(Tsrf1)
SR2;SR2(Tsrf2)
SR3;SR3(Tsrf3)
SR4;SR4(Tsrf4)
SR5;SR5(Tsrf5)
KEEPER
KEEPER;KEEPER(Tnpcf)
SR6;SR6(Tsrf6)
SR7;SR7(Tsrf7)
SR8;SR8(Tsrf8)
SR9;SR9(Tsrf9)
KEEPER; RETURN IntFlush
READOUT:
KEEPER; IF Npump CALL PUMP(Npump)
KEEPER; CALL FCLR(RowSkip)
KEEPER; CALL SCLR(PixSkip)
KEEPER; CALL SCLR(PixRead)
KEEPER; CALL SCLR(PixOver)
STARTFRAME; CALL ActiveLine(RowRead1)
KEEPER;CALL DELREAD(ReadDel)
KEEPER; CALL ActiveLine(RowRead2)
KEEPER; CALL FCLR(RowOver)
KEEPER; CALL SCLR(PixSkip)
KEEPER; CALL SCLR(PixRead)
KEEPER; CALL SCLR(PixOver)
KEEPER; Nexpo--
IDLE; RETURN READOUT
DELREAD:
KEEPER; KEEPER (99998)
KEEPER; RETURN DELREAD 
PUMP:
KEEPER; CALL FWD(Pdepth)
KEEPER; CALL BWD(Pdepth)
KEEPER; KEEPER (100000)
KEEPER; RETURN PUMP 
FWD:
PB; PB(Tpb)
PC; PC(Tpu)
PC; PC(Tpu)
PD; PD(Tpb)
PE; PE(Tpu)
PE; RETURN FWD 
BWD:
PE; PE(Tpu)
PD; PD(Tpb)
PC; PC(Tpu)
PC; PC(Tpu)
PB; PB(Tpb)
PA; RETURN BWD 
ActiveLine:
KEEPER; CALL ParallelShift
KEEPER; CALL SCLR(PixSkip)
RGLOW;
STARTLINE; CALL ReadPixel(PixRead)
KEEPER; CALL SkipPixel
KEEPER; CALL SCLR(PixOver)
KEEPER; KEEPER(Twait)
KEEPER; RETURN ActiveLine
ReadPixel:
SR1;SR1(Tsr1)
SR2;SR2(Tsr2)
SR3;SR3(Tsr3)
SR4;SR4(Tsr4)
SR5;SR5(Tsr5)
PCLK
NOPCLK;NOPCLK(Tnpc)
SR6;SR6(Tsr6)
SR7;SR7(Tsr7)
SR8;SR8(Tsr8)
SR9;SR9(Tsr9)
KEEPER; RETURN ReadPixel
SkipPixel:
SR1;SR1(Tsr1)
SR2;SR2(Tsr2)
SR3;SR3(Tsr3)
SR4;SR4(Tsr4)
SR5;SR5(Tsr5)
KEEPER
KEEPER;KEEPER(Tnpc)
SR6;SR6(Tsr6)
SR7;SR7(Tsr7)
SR8;SR8(Tsr8)
SR9;SR9(Tsr9)
KEEPER; RETURN SkipPixel
ParallelShift:
PA; PA(Tpa)
PB; PB(Tpb)
PC; PC(Tpb)
PD; PD(Tpb)
PE; PE(Tpb)
PF; PF(Tpc)
PI; RETURN ParallelShift
